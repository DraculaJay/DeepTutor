cmake_minimum_required(VERSION 3.24)
set(CMAKE_OSX_DEPLOYMENT_TARGET "14.2" CACHE STRING "Core Audio process tap minimum")
project(DeskmateNative LANGUAGES CXX OBJCXX)
if(NOT APPLE)
  message(FATAL_ERROR "The Deskmate capture adapter targets macOS")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD 20)
option(JARVIS_RUNTIME_ENABLE_UPSTREAM "Build the optional pinned Omni runtime (no weights downloaded)" OFF)
set(JARVIS_REFERENCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../pub-local-jarvis" CACHE PATH "Read-only Jarvis reference checkout")
set(JARVIS_RUNTIME_UPSTREAM_SOURCE_DIR "${JARVIS_REFERENCE_ROOT}/third_party/runtime/vendor" CACHE PATH "Pinned, already patched upstream")
add_subdirectory(third_party/runtime)
add_library(jarvis_worker_core STATIC src/protocol.cpp src/audio.cpp src/fingerprint.cpp src/scheduler.cpp src/runtime.cpp src/worker.cpp)
target_include_directories(jarvis_worker_core PUBLIC include "${JARVIS_RUNTIME_UPSTREAM_SOURCE_DIR}/vendor")
target_link_libraries(jarvis_worker_core PUBLIC jarvis::runtime_boundary)
add_executable(deskmate-native-worker src/macos/main.mm src/macos/server.cpp src/macos/capture.mm src/macos/host_runtime.cpp)
set_source_files_properties(src/macos/capture.mm src/macos/main.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc -fblocks")
target_link_libraries(deskmate-native-worker PRIVATE jarvis_worker_core "-framework AppKit" "-framework ScreenCaptureKit" "-framework CoreAudio" "-framework CoreMedia" "-framework CoreVideo")
if(JARVIS_RUNTIME_ENABLE_UPSTREAM)
  target_sources(jarvis_worker_core PRIVATE src/omni_runtime.cpp)
  target_link_libraries(jarvis_worker_core PRIVATE jarvis::runtime_provider)
  target_compile_definitions(deskmate-native-worker PRIVATE JARVIS_HAS_REAL_RUNTIME=1)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(jarvis-native-tests tests/native_tests.cpp src/macos/capture.mm src/macos/host_runtime.cpp)
  target_link_libraries(jarvis-native-tests PRIVATE jarvis_worker_core "-framework AppKit" "-framework ScreenCaptureKit" "-framework CoreAudio" "-framework CoreMedia" "-framework CoreVideo")
  add_test(NAME jarvis-native-tests COMMAND jarvis-native-tests)
endif()
install(TARGETS deskmate-native-worker RUNTIME DESTINATION bin)
